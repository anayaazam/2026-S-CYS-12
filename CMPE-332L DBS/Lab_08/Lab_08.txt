DROP PROCEDURE IF EXISTS CheckEmployeeAge;
DELIMITER //
CREATE PROCEDURE CheckEmployeeAge(IN BirthYear INT)
BEGIN
    DECLARE age INT;
    SET age = YEAR(CURDATE()) - BirthYear;
    IF age > 18 THEN
        SELECT 'Adult' AS Category;
    ELSE
        SELECT 'Minor' AS Category;
    END IF;
END //
DELIMITER ;
-- Test with BirthYear for age 25
CALL CheckEmployeeAge(YEAR(CURDATE()) - 25);
-- Test with BirthYear for age 16
CALL CheckEmployeeAge(YEAR(CURDATE()) - 16);

-- QUERY 5:
DROP PROCEDURE IF EXISTS AddProductUnits;
DELIMITER //
CREATE PROCEDURE AddProductUnits(
    IN p_ProductID INT,
    IN p_ExtraUnits INT,
    OUT p_TotalStock INT
)
BEGIN
    DECLARE currentStock INT;
    SELECT COALESCE(SUM(CASE WHEN transaction_type = 1 THEN quantity ELSE -quantity END), 0)
    INTO currentStock
    FROM inventory_transactions
    WHERE product_id = p_ProductID;
    SET p_TotalStock = currentStock + p_ExtraUnits;
END //
DELIMITER ;
-- Example execution
CALL AddProductUnits(3, 50, @newStock);
SELECT @newStock AS CalculatedStock;

DELIMITER //
CREATE PROCEDURE OrderCategory(IN p_Freight DECIMAL(10,2))
BEGIN
    SELECT
        p_Freight AS FreightAmount,
        CASE
            WHEN p_Freight >= 100 THEN 'High Freight'
            WHEN p_Freight >= 50 AND p_Freight < 100 THEN 'Medium Freight'
            WHEN p_Freight < 50 THEN 'Low Freight'
        END AS FreightCategory;
END //
DELIMITER ;
-- Example execution
CALL OrderCategory(75.50);

-- QUERY 8:
SELECT
    s.id AS ProductID,
    s.product_name AS ProductName,
    s.units_in_stock AS UnitsInStock,
    CASE
        WHEN s.units_in_stock = 0 THEN 'Out of Stock'
        WHEN s.units_in_stock BETWEEN 1 AND 20 THEN 'Low Stock'
        WHEN s.units_in_stock BETWEEN 21 AND 50 THEN 'Medium Stock'
        WHEN s.units_in_stock > 50 THEN 'High Stock'
    END AS StockStatus
FROM (
    SELECT p.id, p.product_name,
           COALESCE(SUM(CASE WHEN t.transaction_type = 1 THEN t.quantity ELSE -t.quantity END), 0) AS units_in_stock
    FROM products p
    LEFT JOIN inventory_transactions t ON t.product_id = p.id
    GROUP BY p.id, p.product_name
) AS s;

DROP PROCEDURE IF EXISTS ListEmployeeOrders;
DELIMITER //
CREATE PROCEDURE ListEmployeeOrders(IN p_EmployeeID INT)
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE v_OrderID INT;
    DECLARE v_OrderDate DATE;
    -- Declare cursor to fetch orders
    DECLARE order_cursor CURSOR FOR
        SELECT id, order_date
        FROM orders
        WHERE employee_id = p_EmployeeID;
    -- Declare handler for end of data
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
    -- Create a temporary table to display multiple rows from a procedure loop
    DROP TEMPORARY TABLE IF EXISTS TempEmployeeOrders;
    CREATE TEMPORARY TABLE TempEmployeeOrders (OrderID INT, OrderDate DATE);
    OPEN order_cursor;
    read_loop: LOOP
        FETCH order_cursor INTO v_OrderID, v_OrderDate;
        IF done THEN
            LEAVE read_loop;
        END IF;
        INSERT INTO TempEmployeeOrders VALUES (v_OrderID, v_OrderDate);
    END LOOP;
    CLOSE order_cursor;
    -- Display the results
    SELECT * FROM TempEmployeeOrders;
END //
DELIMITER ;
-- Example execution
CALL ListEmployeeOrders(1);

DROP PROCEDURE IF EXISTS CustomerCategory;
DELIMITER //
CREATE PROCEDURE CustomerCategory(IN p_CustomerID INT)
BEGIN
    DECLARE total_orders INT;
    SELECT COUNT(*) INTO total_orders
    FROM orders
    WHERE customer_id = p_CustomerID;
    SELECT
        total_orders AS TotalOrders,
        CASE
            WHEN total_orders >= 50 THEN 'Top Customer'
            WHEN total_orders >= 10 AND total_orders < 50 THEN 'Regular'
            WHEN total_orders < 10 THEN 'Occasional'
        END AS CustomerType;
END //
DELIMITER ;
-- Example execution
CALL CustomerCategory(1);
