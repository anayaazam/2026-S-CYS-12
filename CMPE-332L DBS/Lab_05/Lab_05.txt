-- QUERY 1:
SELECT country_region AS Country, COUNT(id) AS Total_Customers
FROM customers
GROUP BY country_region;
***
 
 
-- QUERY 2:
SELECT category AS CategoryName, COUNT(id) AS Total_Products
FROM products
GROUP BY category;
***
 
 
-- QUERY 3:
SELECT e.id AS EmployeeID, e.last_name AS LastName, COUNT(o.id) AS Total_Orders
FROM employees e
JOIN orders o ON e.id = o.employee_id
GROUP BY e.id, e.last_name;
***
 
 
-- QUERY 4:
SELECT p.id AS ProductID, p.product_name AS ProductName, SUM(od.quantity) AS Total_Quantity
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.id, p.product_name;
***
 
 
-- QUERY 5:
SELECT s.id AS ShipperID, s.company AS ShipperName, COUNT(o.id) AS Total_Orders
FROM shippers s
JOIN orders o ON s.id = o.shipper_id
GROUP BY s.id, s.company;
***
 
 
-- QUERY 6:
SELECT c.id AS CustomerID, c.company AS CustomerName, COUNT(o.id) AS Total_Orders
FROM customers c
JOIN orders o ON c.id = o.customer_id
GROUP BY c.id, c.company
HAVING COUNT(o.id) > 5;
***
 
 
-- QUERY 7:
SELECT p.id AS ProductID, p.product_name AS ProductName, AVG(od.quantity) AS Avg_Quantity
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.id, p.product_name;
***
 
 
-- QUERY 8:
SELECT p.id AS ProductID, p.product_name AS ProductName, SUM(od.quantity) AS Total_Quantity
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.id, p.product_name
ORDER BY Total_Quantity DESC;
***
 
 
-- QUERY 9:
SELECT MONTH(order_date) AS OrderMonth, COUNT(id) AS Total_Orders
FROM orders
GROUP BY MONTH(order_date)
ORDER BY OrderMonth;
***
 
 
-- QUERY 10:
SELECT p.category AS CategoryName, SUM(od.quantity) AS Total_SalesQuantity
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.category;
***
 
 
-- QUERY 11:
SELECT e.id AS EmployeeID, e.last_name AS LastName, COUNT(o.id) AS Total_Orders
FROM employees e
JOIN orders o ON e.id = o.employee_id
GROUP BY e.id, e.last_name
HAVING COUNT(o.id) > 20;
***
 
 
-- QUERY 12:
SELECT p.id AS ProductID, p.product_name AS ProductName, SUM(od.quantity) AS Total_Quantity
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.id, p.product_name
HAVING SUM(od.quantity) > 50;
***
 
 
-- QUERY 13:
SELECT YEAR(order_date) AS OrderYear, COUNT(id) AS Total_Orders
FROM orders
GROUP BY YEAR(order_date)
ORDER BY OrderYear;
***
 
 
-- QUERY 14:
SELECT e.id AS EmployeeID, e.last_name AS LastName, COUNT(o.id) AS Total_Orders
FROM employees e
LEFT JOIN orders o ON e.id = o.employee_id
GROUP BY e.id, e.last_name;
***
 
 
-- QUERY 15:
SELECT p.id AS ProductID, p.product_name AS ProductName
FROM products p
LEFT JOIN order_details od ON p.id = od.product_id
WHERE od.product_id IS NULL;
***
 
 
-- QUERY 16:
SELECT c.country_region AS Country, COUNT(o.id) AS Total_Orders
FROM customers c
JOIN orders o ON c.id = o.customer_id
GROUP BY c.country_region;
***
 
 
-- QUERY 17:
SELECT e.id AS EmployeeID, e.last_name AS LastName, COUNT(DISTINCT od.product_id) AS Unique_Products_Count
FROM employees e
JOIN orders o ON e.id = o.employee_id
JOIN order_details od ON o.id = od.order_id
GROUP BY e.id, e.last_name;
