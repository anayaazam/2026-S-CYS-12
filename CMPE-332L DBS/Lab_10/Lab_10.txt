-- QUERY 2:
CREATE TABLE customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL
);
***
 
 
-- QUERY 3:
CREATE TABLE categories (
    id INT AUTO_INCREMENT PRIMARY KEY,
    category_name VARCHAR(50) NOT NULL
);
***
 
 
-- QUERY 4:
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    product_name VARCHAR(100) NOT NULL,
    category_id INT,
    price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (category_id) REFERENCES categories(id)
);
***
 
 
-- QUERY 5:
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    order_date DATE NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (customer_id) REFERENCES customers(id)
);
***
 
 
-- QUERY 6:
CREATE TABLE order_details (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT,
    product_id INT,
    quantity INT NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id),
    FOREIGN KEY (product_id) REFERENCES products(id)
);
***

DELIMITER //
CREATE PROCEDURE generate_bulk_data()
BEGIN
    DECLARE i INT DEFAULT 1;
    DECLARE rand_cust INT;
    DECLARE rand_prod INT;
    DECLARE rand_qty INT;
    DECLARE rand_price DECIMAL(10,2);
    -- 1. Insert Categories
    INSERT INTO categories (category_name) VALUES ('Electronics'), ('Clothing'), ('Home'), ('Books'), ('Toys');
    -- 2. Insert 50,000 Customers
    WHILE i <= 50000 DO
        INSERT INTO customers (customer_name, email)
        VALUES (CONCAT('Customer ', i), CONCAT('customer', i, '@example.com'));
        SET i = i + 1;
    END WHILE;
    -- 3. Insert 1,000 Products
    SET i = 1;
    WHILE i <= 1000 DO
        INSERT INTO products (product_name, category_id, price)
        VALUES (CONCAT('Product ', i), FLOOR(1 + RAND() * 5), ROUND(10 + RAND() * 500, 2));
        SET i = i + 1;
    END WHILE;
    -- 4. Insert 50,000 Orders
    SET i = 1;
    WHILE i <= 50000 DO
        SET rand_cust = FLOOR(1 + RAND() * 50000);
        INSERT INTO orders (customer_id, order_date, total_amount)
        VALUES (rand_cust, DATE_SUB(CURDATE(), INTERVAL FLOOR(RAND() * 365) DAY), ROUND(50 + RAND() * 1000, 2));
        SET i = i + 1;
    END WHILE;
    -- 5. Insert 50,000 order_details
    SET i = 1;
    WHILE i <= 50000 DO
        SET rand_prod = FLOOR(1 + RAND() * 1000);
        SET rand_qty = FLOOR(1 + RAND() * 5);
        INSERT INTO order_details (order_id, product_id, quantity)
        VALUES (i, rand_prod, rand_qty);
        SET i = i + 1;
    END WHILE;
END //
DELIMITER ;

CREATE VIEW complete_order_view AS
SELECT
    o.id AS order_id,
    o.order_date,
    c.customer_name,
    p.product_name,
    od.quantity,
    p.price
FROM orders o
JOIN customers c ON o.customer_id = c.id
JOIN order_details od ON o.id = od.order_id
JOIN products p ON od.product_id = p.id;

-- QUERY 22:
CREATE TABLE materialized_sales_per_product AS
SELECT
    p.id AS product_id,
    p.product_name,
    SUM(od.quantity * p.price) AS total_sales
FROM products p
JOIN order_details od ON p.id = od.product_id
GROUP BY p.id, p.product_name;
***
 
 
-- QUERY 23:
CREATE TABLE materialized_orders_per_customer AS
SELECT
    c.id AS customer_id,
    c.customer_name,
    COUNT(o.id) AS total_orders
FROM customers c
JOIN orders o ON c.id = o.customer_id
GROUP BY c.id, c.customer_name;
