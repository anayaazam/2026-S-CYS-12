CREATE TABLE countries (
    code INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    continent_name VARCHAR(100) NULL
);

***

CREATE TABLE users (
    id INT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL CHECK (email LIKE '%@%.%'),
    gender VARCHAR(10),
    date_of_birth VARCHAR(20),
    country_code INT,
    created_at VARCHAR(50) DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (country_code) REFERENCES countries(code)
);

***

CREATE TABLE merchants (
    id INT PRIMARY KEY,
    merchant_name VARCHAR(100) NOT NULL,
    admin_id INT,
    country_code INT,
    created_at VARCHAR(50) DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (admin_id) REFERENCES users(id),
    FOREIGN KEY (country_code) REFERENCES countries(code)
);

***

CREATE TABLE products (
    id INT PRIMARY KEY,
    merchant_id INT,
    name VARCHAR(100) NOT NULL,
    price INT NOT NULL,
    status VARCHAR(2) NOT NULL CHECK (status IN ('A', 'NA')),
    created_at VARCHAR(50) DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (merchant_id) REFERENCES merchants(id)
);

***

CREATE TABLE orders (
    id INT PRIMARY KEY,
    user_id INT,
    status VARCHAR(50),
    created_at VARCHAR(50) DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

***

CREATE TABLE order_items (
    order_id INT,
    product_id INT,
    quantity INT NOT NULL CHECK (quantity >= 50),
    PRIMARY KEY (order_id, product_id),
    FOREIGN KEY (order_id) REFERENCES orders(id),
    FOREIGN KEY (product_id) REFERENCES products(id)
);

***

DELIMITER //

CREATE TRIGGER trg_prevent_na_product_order
BEFORE INSERT ON order_items
FOR EACH ROW
BEGIN
    DECLARE prod_status VARCHAR(2);
    SELECT status INTO prod_status FROM products WHERE id = NEW.product_id;
    
    IF prod_status = 'NA' THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Cannot order a product that is Not Available (NA).';
    END IF;
END //

DELIMITER ;
